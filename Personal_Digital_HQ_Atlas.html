el.addEventListener('mousemove', (e) => {
  const r = el.getBoundingClientRect();
  const x = (e.clientX - r.left - r.width / 2) * 0.35;
  const y = (e.clientY - r.top - r.height / 2) * 0.35;
  el.style.transform = `translate(${x}px, ${y}px)`;

CQRS Handler Örneği

C# (.NET 10)
public record GetProjectBySlugQuery(string Slug) : IRequest<Result<ProjectDetailDto>>;

public class GetProjectBySlugQueryHandler : IRequestHandler<GetProjectBySlugQuery, Result<ProjectDetailDto>>
{
    private readonly IApplicationDbContext _context;
    private readonly IHybridCache _cache;

    public async Task<Result<ProjectDetailDto>> Handle(GetProjectBySlugQuery request, CancellationToken ct)
    {
        var project = await _cache.GetOrCreateAsync($"project_{request.Slug}", async _ =>
            await _context.Projects.AsNoTracking()
                .Where(p => p.Slug == request.Slug)
                .Select(p => new ProjectDetailDto(p.Id, p.Title, p.Summary, p.CaseStudyContent))
                .FirstOrDefaultAsync(ct), cancellationToken: ct);

        return project is null ? Result.Failure<ProjectDetailDto>(Errors.NotFound) : Result.Success(project);
    }
}
